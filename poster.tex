\documentclass[landscape]{jlreq}
\usepackage{luacode}
\usepackage{graphicx}
\usepackage[absolute]{textpos}

\fboxsep=0pt
\newcommand{\includepdf}[2]{\begin{textblock*}{0.93\pagewidth}(1pt,1pt)
    \noindent\fbox{\includegraphics[page=#1,width=\columnwidth]{#2}}
\end{textblock*}}

\begin{document}
\pagestyle{empty}
\begin{luacode*}
local path = 'output.pdf'
local pages = pdfe.getnofpages(pdfe.open(path))
for i = 1, pages do
    tex.print('\\includepdf{' .. i .. '}{' .. path .. '}')
    tex.print('\\null\\clearpage')
end
\end{luacode*}
\end{document}
